cmake_minimum_required(VERSION 3.16)
project(ultimaxb C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(SDL2 CONFIG REQUIRED)
add_executable(ultimaxb src/main.c)
target_link_libraries(ultimaxb PRIVATE SDL2::SDL2)
target_compile_options(ultimaxb PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME renderer_smoke COMMAND ultimaxb --smoke)
set_tests_properties(renderer_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
